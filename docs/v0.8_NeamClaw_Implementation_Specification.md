#### 3.3.2 `parse_claw_agent()`

```cpp
StmtPtr Parser::parse_claw_agent(const Visibility& visibility)
{
  SourceSpan span = span_from_token(previous());
  if (!match(TokenType::Identifier))
  {
    error("Expected claw agent name");
  }
  const auto name = previous().lexeme;
  if (!match(TokenType::LeftBrace))
  {
    error("Expected '{' after claw agent name");
  }

  ClawAgentDecl decl;
  decl.visibility = visibility;
  decl.name = name;

  while (!check(TokenType::RightBrace) && !is_at_end())
  {
    // Common fields (reuse existing parse patterns from parse_agent)
    if (match(TokenType::Provider))
    {
      expect_colon_string("provider");
      decl.provider = previous().lexeme;
      continue;
    }
    if (match(TokenType::Model))
    {
      expect_colon_string("model");
      decl.model = previous().lexeme;
      continue;
    }
    if (match(TokenType::System))
    {
      expect_colon_string("system");
      decl.system = previous().lexeme;
      continue;
    }
    // ... (endpoint, api_key_env, temperature — same pattern as parse_agent)

    // Claw-specific: session: { ... }
    if (match_identifier("session"))
    {
      expect(TokenType::Colon, "Expected ':' after session");
      decl.session = parse_session_config();
      continue;
    }

    // Claw-specific: channels: [...]
    if (match_identifier("channels"))
    {
      expect(TokenType::Colon, "Expected ':' after channels");
      decl.channels = parse_identifier_list();
      continue;
    }

    // Claw-specific: lanes: { ... }
    if (match_identifier("lanes"))
    {
      expect(TokenType::Colon, "Expected ':' after lanes");
      decl.lanes = parse_lane_configs();
      continue;
    }

    // Claw-specific: semantic_memory: { ... }
    if (match_identifier("semantic_memory"))
    {
      expect(TokenType::Colon, "Expected ':' after semantic_memory");
      decl.semantic_memory = parse_semantic_memory_config();
      continue;
    }

    // Skills, guardchains, budget, policy, workspace — same as parse_agent
    if (match(TokenType::Skills)) { /* ... */ continue; }
    if (match(TokenType::GuardChains)) { /* ... */ continue; }
    if (match(TokenType::Budget)) { /* ... */ continue; }
    if (match(TokenType::Policy)) { /* ... */ continue; }

    // Workspace: string value
    if (match_identifier("workspace"))
    {
      expect_colon_string("workspace");
      decl.workspace = previous().lexeme;
      continue;
    }

    // FIELD REJECTION: reject forge-only fields with clear error
    if (match_identifier("verify"))
    {
      error("'verify' is not valid on a claw agent. Use 'forge agent' for verification-driven loops.");
    }
    if (match_identifier("checkpoint"))
    {
      error("'checkpoint' is not valid on a claw agent. Use 'forge agent' for checkpointed loops.");
    }
    if (match_identifier("loop"))
    {
      error("'loop' is not valid on a claw agent. Use 'forge agent' for iteration loops.");
    }

    // DEPRECATION WARNING: mode field
    if (match_identifier("mode"))
    {
      warning("'mode' field is deprecated in v0.8. Use 'claw agent' or 'forge agent' keywords instead.");
      expect(TokenType::Colon, "Expected ':' after mode");
      advance(); // consume value
      continue;
    }

    // Unknown field
    error("Unexpected field in claw agent declaration: " + current().lexeme);
  }

  expect(TokenType::RightBrace, "Expected '}' after claw agent body");

  // REQUIRED FIELD VALIDATION
  if (decl.provider.empty()) error("claw agent '" + name + "' missing required field 'provider'");
  if (decl.model.empty()) error("claw agent '" + name + "' missing required field 'model'");
  if (decl.channels.empty()) error("claw agent '" + name + "' missing required field 'channels'");
  // session has defaults, so not strictly required to specify every sub-field

  auto stmt = std::make_unique<Statement>();
  stmt->span = span;
  stmt->node = std::move(decl);
  return stmt;
}
```

```cpp
StmtPtr Parser::parse_forge_agent(const Visibility& visibility)
{
  SourceSpan span = span_from_token(previous());
  if (!match(TokenType::Identifier))
  {
    error("Expected forge agent name");
  }
  const auto name = previous().lexeme;
  if (!match(TokenType::LeftBrace))
  {
    error("Expected '{' after forge agent name");
  }

  ForgeAgentDecl decl;
  decl.visibility = visibility;
  decl.name = name;

  while (!check(TokenType::RightBrace) && !is_at_end())
  {
    // Common fields
    if (match(TokenType::Provider)) { /* ... */ continue; }
    if (match(TokenType::Model)) { /* ... */ continue; }
    if (match(TokenType::System)) { /* ... */ continue; }

    // Forge-specific: loop: { ... }
    if (match_identifier("loop"))
    {
      expect(TokenType::Colon, "Expected ':' after loop");
      decl.loop = parse_loop_config();
      continue;
    }

    // Forge-specific: verify: fun(ctx) { ... }
    if (match_identifier("verify"))
    {
      expect(TokenType::Colon, "Expected ':' after verify");
      decl.verify = parse_expression();
      continue;
    }

    // Forge-specific: checkpoint: "git"
    if (match_identifier("checkpoint"))
    {
      expect_colon_string("checkpoint");
      decl.checkpoint = previous().lexeme;
      continue;
    }

    // Skills, guardchains, budget, policy, workspace
    if (match(TokenType::Skills)) { /* ... */ continue; }
    if (match(TokenType::GuardChains)) { /* ... */ continue; }
    if (match(TokenType::Budget)) { /* ... */ continue; }
    if (match(TokenType::Policy)) { /* ... */ continue; }
    if (match_identifier("workspace"))
    {
      expect_colon_string("workspace");
      decl.workspace = previous().lexeme;
      continue;
    }

    // FIELD REJECTION: reject claw-only fields
    if (match_identifier("session"))
    {
      error("'session' is not valid on a forge agent. Use 'claw agent' for session management.");
    }
    if (match_identifier("channels"))
    {
      error("'channels' is not valid on a forge agent. Use 'claw agent' for multi-channel support.");
    }
    if (match_identifier("lanes"))
    {
      error("'lanes' is not valid on a forge agent. Use 'claw agent' for lane queues.");
    }

    // DEPRECATION WARNING
    if (match_identifier("mode"))
    {
      warning("'mode' field is deprecated in v0.8. Use 'claw agent' or 'forge agent' keywords instead.");
      expect(TokenType::Colon, "Expected ':' after mode");
      advance();
      continue;
    }

    error("Unexpected field in forge agent declaration: " + current().lexeme);
  }

  expect(TokenType::RightBrace, "Expected '}' after forge agent body");

  // REQUIRED FIELD VALIDATION
  if (decl.provider.empty()) error("forge agent '" + name + "' missing required field 'provider'");
  if (decl.model.empty()) error("forge agent '" + name + "' missing required field 'model'");
  if (!decl.verify) error("forge agent '" + name + "' missing required field 'verify'");

  auto stmt = std::make_unique<Statement>();
  stmt->span = span;
  stmt->node = std::move(decl);
  return stmt;
}
```

```cpp
// Match against an identifier lexeme (not a keyword token)
bool Parser::match_identifier(const std::string& name)
{
  if (check(TokenType::Identifier) && current().lexeme == name)
  {
    advance();
    return true;
  }
  return false;
}

// Expect colon then string literal
void Parser::expect_colon_string(const std::string& field_name)
{
  if (!match(TokenType::Colon))
    error("Expected ':' after " + field_name);
  if (!match(TokenType::String))
    error("Expected string literal for " + field_name);
}

// Expect colon then number literal
void Parser::expect_colon_number(const std::string& field_name)
{
  if (!match(TokenType::Colon))
    error("Expected ':' after " + field_name);
  if (!match(TokenType::Number))
    error("Expected number literal for " + field_name);
}

// Consume optional trailing comma between fields
void Parser::consume_optional_comma()
{
  match(TokenType::Comma);
}
```

#### 4.2.1 `emit_claw_agent()`

```cpp
void Compiler::emit_claw_agent(const ClawAgentDecl& decl)
{
  // Push agent name
  emit_constant(vm::Value::String(decl.name.c_str(), decl.name.size()));

  // Push common fields as name/value pairs
  uint8_t field_count = 0;

  auto push_string_field = [&](const std::string& name, const std::string& value) {
    emit_constant(vm::Value::String(name.c_str(), name.size()));
    emit_constant(vm::Value::String(value.c_str(), value.size()));
    field_count++;
  };

  push_string_field("provider", decl.provider);
  push_string_field("model", decl.model);
  if (decl.system) push_string_field("system", *decl.system);
  if (decl.workspace) push_string_field("workspace", *decl.workspace);

  // Push session config as a nested object
  emit_constant(vm::Value::String("session", 7));
  emit_session_config(decl.session);
  field_count++;

  // Push channels as a list
```

```cpp
bool Compiler::is_neamclaw_trait(const std::string& name) const
{
  static const std::unordered_set<std::string> traits = {
    "Channelable", "Schedulable", "Sandboxable",
    "Monitorable", "Orchestrable", "Searchable"
  };
  return traits.count(name) > 0;
}

void Compiler::validate_neamclaw_trait_compat(const std::string& trait, const std::string& type)
{
  // Look up agent type (requires tracking agent declarations during compilation)
  auto it = agent_types_.find(type);
  if (it == agent_types_.end()) return; // not an agent, let it pass

  AgentKind kind = it->second;

  // Channelable and Schedulable: claw only
  if ((trait == "Channelable" || trait == "Schedulable") && kind != AgentKind::Claw)
  {
    error("trait '" + trait + "' can only be implemented for claw agents, but '" +
          type + "' is a " + agent_kind_name(kind) + " agent");
  }

  // Orchestrable: claw and forge only
  if (trait == "Orchestrable" && kind == AgentKind::Stateless)
  {
    warning("trait '" + trait + "' on stateless agent '" + type + "' is unusual");
  }
}
```

```cpp
LoopOutcome ForgeController::run(ObjForgeAgent* agent, const std::string& workspace)
{
  const auto& loop = agent->loop;
  int max_iter = loop.max_iterations;

  for (int i = 1; i <= max_iter; ++i)
  {
    // 1. Create fresh loop context
    auto* ctx = allocate_object<ObjLoopContext>(vm_);
    ctx->type = OBJ_LOOP_CONTEXT;
    ctx->agent = agent;
    ctx->iteration = i;

    // 2. Select next task from plan file
    if (loop.plan_file)
    {
      ctx->current_task = select_next_task(*loop.plan_file);
      if (ctx->current_task.empty())
      {
        return {LoopOutcomeKind::Completed, i, "All tasks complete", ctx->total_cost};
      }
    }

    // 3. Execute iteration (fresh VM from pool if available)
    execute_iteration(agent, ctx);

    // 4. Run verification
    Value verify_result = run_verify(agent, ctx);

    // 5. Process verify result (VerifyResult sealed type)
    std::string variant_name = get_variant_name(verify_result);
    if (variant_name == "Done")
    {
      checkpoint(agent, i);
      append_learnings(loop.learnings_file.value_or(""), i,
                       ctx->current_task, "completed");
      continue; // next task
    }
    else if (variant_name == "Retry")
    {
      ctx->feedback = get_variant_data(verify_result);
      // feedback injected in next iteration
      continue;
    }
    else if (variant_name == "Abort")
    {
      return {LoopOutcomeKind::Aborted, i, get_variant_data(verify_result), ctx->total_cost};
    }

    // 6. Check budget
    if (loop.max_cost > 0 && ctx->total_cost >= loop.max_cost)
    {
      return {LoopOutcomeKind::BudgetExhausted, i, "Cost limit exceeded", ctx->total_cost};
    }
  }

  return {LoopOutcomeKind::MaxIterations, max_iter, "Hit iteration limit", 0.0};
}
```

```cpp
    Value interval = invoke_trait_method(agent->name, "Schedulable",
                                         "heartbeat_interval", self, {});
    if (interval.is_number()) interval_minutes = static_cast<int>(interval.as_number());
  }

  while (!g_draining.load())  // v0.7.2 graceful shutdown
  {
    std::this_thread::sleep_for(std::chrono::minutes(interval_minutes));
    if (g_draining.load()) break;

    Value result = invoke_trait_method(agent->name, "Schedulable",
                                       "on_heartbeat", self, {});
    // Process HeartbeatAction result...
    std::string variant = get_variant_name(result);
    if (variant == "Notify")
    {
      // Route notification through channels
      std::string message = get_variant_data(result);
      route_heartbeat_notification(agent, message);
    }
    // "Silent" — do nothing
  }
}
```

### 10.3 SQLite Integration

The memory index uses SQLite (already available in most environments via system library
or can be bundled). The database schema:

```sql
CREATE TABLE IF NOT EXISTS chunks (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  file_path TEXT NOT NULL,
  chunk_index INTEGER NOT NULL,
  content TEXT NOT NULL,
  updated_at REAL NOT NULL
);

CREATE VIRTUAL TABLE IF NOT EXISTS chunks_fts USING fts5(content, content='chunks', content_rowid='id');
```

For the initial implementation, semantic search uses BM25 via FTS5 (keyword matching).
Vector similarity can be added later using a lightweight embedding model or external
service. The `search` parameter controls the mode:
- `"keyword"` — FTS5 only
- `"hybrid"` — FTS5 + TF-IDF cosine similarity (weighted 30%/70%)
- `"vector"` — future: requires embedding model

```cpp
  // 1. Parse CLI arguments
  std::string agent_file;

  // 2. Compile the agent file
```
